PSPプログラミングその２

また、PSPと戯れてしまったので、その顛末を記述します。


■終了関連
普通のゲームのように「HOME」ボタンで終了が出来るようになった模様。

これで、強制電源Offしなくてすみます。(^-^;A
参考サイトは、こちら。


●startup.s
基本的には、下記の関数のポインタを記述すればOKのはずです。

STUB_START "ThreadManForUser",0x40010000,0x000B0005
STUB_FUNC 0x82826F70,KernelPollCallbacks
STUB_FUNC 0xE81CAF8F,sceKernelCreateCallback
STUB_END

STUB_START "LoadExecForUser",0x40010000,0x20005
STUB_FUNC 0x5572A5F,sceKernelExitGame
STUB_FUNC 0x4AC57943,SetExitCallback
STUB_END

STUB_START "scePower",0x40010000,0x10005
STUB_FUNC 0x4B7766E,PowerSetCallback
STUB_END

●pg.h
下記の物を追加します。
終了のみなので、あまり数はありません。

void sceKernelExitGame(void);
void SetExitCallback(int cbid);
void PowerSetCallback(int zero, int cbid);
void KernelPollCallbacks(void);
int sceKernelCreateCallbacks(const char *name, void *func);
int sceKernelCreateThread(const char *name, void *func, int a,int b,int c,int d);

●終了関連の関数
必要な関数は、下記の物となります。
これらを、main.c などにコピペしてください。

int exit_callback(void)
{
// Cleanup the games resources etc (if required)

sceKernelExitGame();

return 0;
}

#define POWER_CB_POWER 0x80000000
#define POWER_CB_HOLDON 0x40000000
#define POWER_CB_STANDBY 0x00080000
#define POWER_CB_RESCOMP 0x00040000
#define POWER_CB_RESUME 0x00020000
#define POWER_CB_SUSPEND 0x00010000
#define POWER_CB_EXT 0x00001000
#define POWER_CB_BATLOW 0x00000100
#define POWER_CB_BATTERY 0x00000080
#define POWER_CB_BATTPOWER 0x0000007F
void power_callback(int unknown, int pwrflags)
{
// Combine pwrflags and the above defined masks
}

// Thread to create the callbacks and then begin polling
int CallbackThread(void *arg)
{
int cbid;

cbid = sceKernelCreateCallback("Exit Callback", exit_callback);
SetExitCallback(cbid);
cbid = sceKernelCreateCallback("Power Callback", power_callback);
PowerSetCallback(0, cbid);

KernelPollCallbacks();
}

/* Sets up the callback thread and returns its thread id */
int SetupCallbacks(void)
{
int thid = 0;

thid = sceKernelCreateThread("update_thread", CallbackThread, 0x11, 0xFA0, 0, 0);
if(thid >= 0)
{
sceKernelStartThread(thid, 0, 0);
}

return thid;
}


●終了処理用コード
初期化が、終わったあたりに、下記の一行を追加するだけです。
これで、「HOME」ボタンを押すと、見慣れた画面が表示されます。

SetupCallbacks();

これを拡張していけば。。。(^-^;A

●終了処理用コード
さきほどの main.c 内、519行目から532行目までが、ファイル用の配列と構造体です。
必要に応じて、作成してください。

#define WAVFILEMAX_BG 8*1024*1024
char wavdata_bg[WAVFILEMAX_BG];
wavout_wavinfo_t wavinfo_bg;


使い方は、下記のような感じで、ファイル名と上記の構造体を渡して、
再生させます。
「0」と「1」があるのですが、BGMのようなループさせたい音とジャンプ音のような
一回のみ非同期で再生したい音というような感じで使い分けると思われます。(^-^;A


「0」のパターン


ret=wavoutLoadWav("op.wav",&wavinfo_bg,wavdata_bg,sizeof(wavdata_bg));
if (ret!=0) halt("op.wav load error");
wavinfo_bg.playloop=1; // ループします。
wavoutStartPlay0(&wavinfo_bg);
　　//終了処理
wavoutWaitEnd0();
wavoutStopPlay0();

「1」のパターン

ret=wavoutLoadWav("op.wav",&wavinfo_bg,wavdata_bg,sizeof(wavdata_bg));
if (ret!=0) halt("op.wav load error");
wavoutStartPlay1(&wavinfo_bg);
//終了処理
wavoutStopPlay1();
